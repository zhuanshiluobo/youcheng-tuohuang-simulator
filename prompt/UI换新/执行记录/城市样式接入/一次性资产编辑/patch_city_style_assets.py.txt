def meta(path):
 p=Path(str(path)+'.meta')
 if not p.exists():p.write_text('fileFormatVersion: 2\nguid: '+uuid.uuid4().hex+'\n',encoding='utf8')
 return re.search('guid: (\w+)',p.read_text())[1]

def remove_comp(a,i):

def clone_comp(a,source,r):
 t=source
 script=re.search(r'm_Script: .*guid: (\w+)',t)[1]
 body=t.split('  m_EditorClassIdentifier:')[1].split('\n',1)[1]
 return a.component(a.go(r),script,body)
