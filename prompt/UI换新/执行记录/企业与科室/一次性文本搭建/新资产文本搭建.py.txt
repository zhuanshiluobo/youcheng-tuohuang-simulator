def guid(p):
    return re.search(r'^guid: (\w+)',Path(str(p)+'.meta').read_text(encoding='utf-8-sig'),re.M)[1]
def meta(p,kind='DefaultImporter'):
    m=Path(str(p)+'.meta')
    if m.exists(): return guid(p)
    g=uuid.uuid4().hex
    m.write_text(f'fileFormatVersion: 2\nguid: {g}\n'+('folderAsset: yes\n' if p.is_dir() else '')+f'{kind}:\n  externalObjects: {{}}\n  userData: \n  assetBundleName: \n  assetBundleVariant: \n',encoding='utf-8')
    return g

class Asset:
    def __init__(self): self.i=1000; self.objs=[]; self.docs=[]
    def id(self): self.i+=1; return self.i
    def node(self,name,parent=None,active=True,amin=(0,0),amax=(1,1),size=(0,0),pivot=(.5,.5)):
        n={'go':self.id(),'rt':self.id(),'name':name,'parent':parent,'active':active,'children':[],'components':[],'amin':amin,'amax':amax,'size':size,'pivot':pivot}
        if parent: parent['children'].append(n)
        self.objs.append(n); return n
    def comp(self,n,name,body):
        id=self.id(); n['components'].append(id)
        self.docs.append(f'--- !u!114 &{id}\nMonoBehaviour:\n'+base+f'  m_GameObject: {ref(n["go"])}\n  m_Enabled: 1\n  m_EditorHideFlags: 0\n  m_Script: {{fileID: 11500000, guid: {scripts[name]}, type: 3}}\n  m_Name: \n  m_EditorClassIdentifier: \n'+body)
        return id
    def renderer(self,n):
        id=self.id();n['components'].append(id);self.docs.append(f'--- !u!222 &{id}\nCanvasRenderer:\n'+base+f'  m_GameObject: {ref(n["go"])}\n  m_CullTransparentMesh: 1\n')
    def image(self,n,sprite=None,color=(1,1,1,1),ray=0,sliced=1):
        self.renderer(n)
        return self.comp(n,'Image',f'  m_Material: {{fileID: 0}}\n  m_Color: {{r: {color[0]}, g: {color[1]}, b: {color[2]}, a: {color[3]}}}\n  m_RaycastTarget: {ray}\n  m_Maskable: 1\n  m_Sprite: {ext(sprite) if sprite else ref(0)}\n  m_Type: {sliced}\n  m_PreserveAspect: 0\n  m_FillCenter: 1\n  m_PixelsPerUnitMultiplier: 2.5\n')
    def le(self,n,w=-1,h=-1,fw=0,fh=0,minw=0,minh=0,ignore=0):
        return self.comp(n,'LayoutElement',f'  m_IgnoreLayout: {ignore}\n  m_MinWidth: {minw}\n  m_MinHeight: {minh}\n  m_PreferredWidth: {w}\n  m_PreferredHeight: {h}\n  m_FlexibleWidth: {fw}\n  m_FlexibleHeight: {fh}\n  m_LayoutPriority: 1\n')
    def group(self,n,vertical=True,gap=8,pad=0,align=0,expandW=0,expandH=0):
        return self.comp(n,'VerticalLayoutGroup' if vertical else 'HorizontalLayoutGroup',f'  m_Padding:\n    m_Left: {pad}\n    m_Right: {pad}\n    m_Top: {pad}\n    m_Bottom: {pad}\n  m_ChildAlignment: {align}\n  m_Spacing: {gap}\n  m_ChildForceExpandWidth: {expandW}\n  m_ChildForceExpandHeight: {expandH}\n  m_ChildControlWidth: 1\n  m_ChildControlHeight: 1\n  m_ChildScaleWidth: 0\n  m_ChildScaleHeight: 0\n  m_ReverseArrangement: 0\n')
    def text(self,n,text='',role='normal',size=22,align=0):
        self.renderer(n)
        return self.comp(n,'Text',f'  m_Material: {{fileID: 0}}\n  m_Color: {{r: 0.95, g: 0.93, b: 0.86, a: 1}}\n  m_RaycastTarget: 0\n  m_Maskable: 1\n  m_FontData:\n    m_Font: {F[role]}\n    m_FontSize: {size}\n    m_FontStyle: 0\n    m_BestFit: 0\n    m_MinSize: 10\n    m_MaxSize: 40\n    m_Alignment: {align}\n    m_AlignByGeometry: 0\n    m_RichText: 0\n    m_HorizontalOverflow: 0\n    m_VerticalOverflow: 1\n    m_LineSpacing: 1\n  m_Text: {scalar(text)}\n')
    def button(self,n,label=None):
        graphic=self.image(n,sprites/'Frontier31/button-secondary.png',ray=1)
        id=self.comp(n,'Button',f'  m_Navigation:\n    m_Mode: 3\n    m_WrapAround: 0\n    m_SelectOnUp: {{fileID: 0}}\n    m_SelectOnDown: {{fileID: 0}}\n    m_SelectOnLeft: {{fileID: 0}}\n    m_SelectOnRight: {{fileID: 0}}\n  m_Transition: 1\n  m_Colors:\n    m_NormalColor: {{r: 1, g: 1, b: 1, a: 1}}\n    m_HighlightedColor: {{r: 1, g: 0.95, b: 0.75, a: 1}}\n    m_PressedColor: {{r: 0.7, g: 0.7, b: 0.7, a: 1}}\n    m_SelectedColor: {{r: 1, g: 0.95, b: 0.75, a: 1}}\n    m_DisabledColor: {{r: 0.4, g: 0.4, b: 0.4, a: 0.6}}\n    m_ColorMultiplier: 1\n    m_FadeDuration: 0.1\n  m_Interactable: 1\n  m_TargetGraphic: {ref(graphic)}\n  m_OnClick:\n    m_PersistentCalls:\n      m_Calls: []\n')
        text=self.text(self.node('Label',n),label,'emphasis',24,4) if label is not None else None
        return id,text
    def raw(self,n):
        self.renderer(n)
        im=self.comp(n,'RawImage','  m_Material: {fileID: 0}\n  m_Color: {r: 1, g: 1, b: 1, a: 1}\n  m_RaycastTarget: 0\n  m_Maskable: 1\n  m_Texture: {fileID: 0}\n  m_UVRect: {x: 0, y: 0, width: 1, height: 1}\n')
        asp=self.comp(n,'AspectRatioFitter','  m_AspectMode: 3\n  m_AspectRatio: 0.35135135\n');return im,asp
    def scroll(self,parent,name,horizontal=False):
        n=self.node(name,parent);self.le(n,fw=1,fh=1); self.image(n,color=(.03,.055,.06,.95),ray=1)
        vp=self.node('Viewport',n);self.comp(vp,'RectMask2D','  m_Padding: {x: 0, y: 0, z: 0, w: 0}\n  m_Softness: {x: 0, y: 0}\n')
        c=self.node('Content',vp,amin=(0,0) if horizontal else (0,1),amax=(0,1) if horizontal else (1,1),pivot=(0,.5) if horizontal else (.5,1))
        self.group(c,not horizontal,gap=34 if horizontal else 12,pad=12,align=4 if horizontal else 0,expandH=1 if horizontal else 0,expandW=0 if horizontal else 1)
        if horizontal:self.comp(c,'UiScrollContentWidth',f'  viewport: {ref(vp["rt"])}\n')
        else:self.comp(c,'ContentSizeFitter','  m_HorizontalFit: 0\n  m_VerticalFit: 2\n')
        sc=self.comp(n,'ScrollRect',f'  m_Content: {ref(c["rt"])}\n  m_Horizontal: {int(horizontal)}\n  m_Vertical: {int(not horizontal)}\n  m_MovementType: 2\n  m_Elasticity: 0.1\n  m_Inertia: 1\n  m_DecelerationRate: 0.135\n  m_ScrollSensitivity: 35\n  m_Viewport: {ref(vp["rt"])}\n  m_HorizontalScrollbar: {{fileID: 0}}\n  m_VerticalScrollbar: {{fileID: 0}}\n  m_HorizontalScrollbarVisibility: 0\n  m_VerticalScrollbarVisibility: 0\n  m_OnValueChanged:\n    m_PersistentCalls:\n      m_Calls: []\n')
        return n,c,sc
    def save(self,p):
        out=header
        for n in self.objs:
            out+=f'--- !u!1 &{n["go"]}\nGameObject:\n'+base+'  serializedVersion: 6\n  m_Component:\n'+''.join(f'  - component: {ref(i)}\n' for i in [n['rt']]+n['components'])+f'  m_Layer: 5\n  m_Name: {scalar(n["name"])}\n  m_TagString: Untagged\n  m_IsActive: {int(n["active"])}\n'
            out+=f'--- !u!224 &{n["rt"]}\nRectTransform:\n'+base+f'  m_GameObject: {ref(n["go"])}\n  m_LocalRotation: {{x: 0, y: 0, z: 0, w: 1}}\n  m_LocalPosition: {{x: 0, y: 0, z: 0}}\n  m_LocalScale: {{x: 1, y: 1, z: 1}}\n  m_Children:'+ ('\n'+''.join(f'  - {ref(c["rt"])}\n' for c in n['children']) if n['children'] else ' []\n')+f'  m_Father: {ref(n["parent"]["rt"] if n["parent"] else 0)}\n  m_AnchorMin: {{x: {n["amin"][0]}, y: {n["amin"][1]}}}\n  m_AnchorMax: {{x: {n["amax"][0]}, y: {n["amax"][1]}}}\n  m_AnchoredPosition: {{x: 0, y: 0}}\n  m_SizeDelta: {{x: {n["size"][0]}, y: {n["size"][1]}}}\n  m_Pivot: {{x: {n["pivot"][0]}, y: {n["pivot"][1]}}}\n'
        new(p,out+''.join(self.docs),'PrefabImporter')
